# The Computer Language Benchmarks Game

import sys
import math

class Node:
    # Define a tree node structure with left and right child pointers
    def __init__(self, value=None):
        self.value = value
        self.left = None
        self.right = None

def stretch_tree(depth):
    # Allocate a binary tree to stretch memory, check it exists,
    #  and deallocate it
    root = Node()
    for i in range(1 << depth):
        root.left = Node(i)
        root = root.left
        if i % 2 == 0:
            root.right = Node(i + 1)
            root = root.right
    del root
    print(f"stretch tree of depth {depth}\t check: C")

def long_lived_tree():
    # Allocate a long-lived binary tree which will live on while
    #  other trees are allocated and deallocated
    root = Node()
    return root

def bottom_up_tree(depth):
    # Allocate, walk, and deallocate many bottom-up binary trees:
    # allocate a tree, walk the tree counting the nodes, deallocate the tree
    root = Node()
    iterations = 1 << (depth - 4)
    for _ in range(iterations):
        root.left = Node()
        root = root.left
        if depth > 4:
            root.right = Node()
            root = root.right
    print(f"I\t trees of depth {depth}\t check: C")
    del root

def main():
    # Receive N as a command-line argument
    n = int(sys.argv[1])

    min_depth = 4
    max_depth = max(min_depth + 2, n)
    stretch_depth = max_depth + 1

    print("stretch tree of depth", stretch_depth, "\t check: C")

    for depth in range(min_depth, max_depth + 1, 2):
        print(f"I\t trees of depth {depth}\t check: C")
        for _ in range(1 << (max_depth - depth + min_depth)):
            root = long_lived_tree()
            bottom_up_tree(depth)
            del root

    print(f"long lived tree of depth {max_depth}\t check: C")

if __name__ == '__main__':
    main(n)
