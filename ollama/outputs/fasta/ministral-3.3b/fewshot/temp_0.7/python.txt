# Example code structure for generating sequences with command-line argument N.
import sys

def calculate_cumulative_probabilities(alphabet):
    """Convert nucleotide probabilities to cumulative distribution."""
    probabilities = {k: v for k, v in alphabet.items()}
    sorted_items = sorted(probabilities.items(), key=lambda x: -x[1])
    cumulative = 0.0
    cumulative_probs = {}
    for nuc, prob in sorted_items:
        cumulative += prob
        cumulative_probs[nuc] = cumulative
    return cumulative_probs

def select_nucleotide(cumulative_prob, random_number):
    """Select a nucleotide using linear search against cumulative probabilities."""
    for nuc, prob in sorted(cumulative_prob.items(), key=lambda x: -x[1]):
        if random_number < prob:
            return nuc
    return list(cumulative_prob.keys())[-1]

def generate_sequence(length, alphabet, seed):
    """Generate sequence by sampling nucleotides using LCG."""
    cumulative_probs = calculate_cumulative_probabilities(alphabet)
    current_seed = seed
    output = []
    for _ in range(length):
        random_number = 0.9 * (current_seed / IM)  # Simplified LCG formula for demo
        nuc = select_nucleotide(cumulative_probs, random_number)
        output.append(nuc)
        current_seed = ((current_seed * IA + IC) % IM)  # LCG step
    return ''.join(output)

if __name__ == "__main__":
    N = int(sys.argv[1])
    ALU_seq = "GGCCGGGCGCGGTGGCTCACGCCTGTAATCCCAGCACTTTGGGAGGCCGAGGCGGGCGGA"  # Example truncated
    seq_len = len(ALU_seq) * 2

    seed = 42
    one_seq = generate_sequence(seq_len, {'a':0.3, 'c':0.197}, seed)
    two_seq = generate_sequence(seq_len, {'a':0.27, 'B':0.02}, seed)  # IUB example (simplified)
    three_seq = generate_sequence(seq_len, {'a':0.30295494, 'c':0.198}, seed)

    print(f">ONE {one_seq[:60]}")
    print(f">TWO {two_seq[:60]}")
    print(f">THREE {three_seq[:60]}")
