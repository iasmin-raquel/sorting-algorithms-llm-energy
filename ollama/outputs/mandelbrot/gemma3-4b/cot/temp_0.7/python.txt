import sys

def mandelbrot(N):
    """
    Generates a Mandelbrot set bitmap image.
    """
    # Header

    lim = 2.0 * 2.0
    for y in range(N):

            real_c = 2.0 * x / N - 1.5
            imag_c = 2.0 * y / N - 1.0

                zi = z.imag
