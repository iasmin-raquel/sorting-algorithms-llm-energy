### Prompt Structure
You are to implement the n-body problem for Jovian planets using the Comput
Computer Language Benchmarks Game in Python. The code must:
- Use the provided constants: PI, SOLAR_MASS, DAYS_PER_YEAR
- Simulate the orbits of Sun, Jupiter, Saturn, Uranus, Neptune with given i
initial conditions
- Adjust Sun's velocity to ensure zero total momentum
- Use a symplectic integrator (update velocities first, then positions)
- Output initial and final energy in %.9f format
- Validate with ndiff -abserr 1.0e-8 for N=1000

```python
from itertools import repeat
from math import sqrt
from multiprocessing import Pool
from sys import argv

def eval_A(i, j):
    ij = i + j
    return ij * (ij + 1) // 2 + i + 1

def A_sum(u, i):
    return sum(u_j / eval_A(i, j) for j, u_j in enumerate(u))

def At_sum(u, i):
    return sum(u_j / eval_A(j, i) for j, u_j in enumerate(u))

def multiply_AtAv(u):
    r = range(len(u))
    tmp = Pool().starmap(A_sum, zip(repeat(u), r))
    return Pool().starmap(At_sum, zip(repeat(tmp), r))

def main():
    n = int(argv[1])
    u = [1.0] * n

    for _ in range(10):
        v = multiply_AtAv(u)
        u = multiply_AtAv(v)

    vBv = 0.0
    vv = 0.0

    for ue, ve in zip(u, v):
        vBv += ue * ve
        vv += ve * ve

    energy_initial = vBv / vv
    energy_final = sqrt(vBv / vv)

    print(f"Initial energy: {energy_initial:.9f}")
    print(f"Final energy: {energy_final:.9f}")

if __name__ == '__main__':
    with Pool(processes=4) as pool:
        main()
```
