```python
# The Computer Language Benchmarks Game

import sys
from sympy import *

N = int(sys.argv[1])
Q, R, T, K = 1, 0, 1, 0

def next_step():
    global Q, R, T, K
    K += 1
    R += 2*Q
    R *= (2*K + 1)
    T *= (2*K + 1)
    Q *= K

def extract(x):
    return floor((Q*x + R) / T)

def produce(d):
    global Q, R, T
    Q *= 10
    R = (R - T*d) * 10

def digit_selection():
    while True:
        next_step()
        d = extract(3)
        if d == extract(4):
            print(f"{d:>{LINELEN}}", end="\t:")
            print(f"{str(N - K//100)[-LINELEN:]}" if (K % 100 != 0) else "\n")
            produce(d)
        else:
            next_step()

if __name__ == '__main__':
    digit_selection()
```
